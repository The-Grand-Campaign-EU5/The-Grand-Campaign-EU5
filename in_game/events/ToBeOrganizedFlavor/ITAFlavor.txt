# Tuscan event
italy.14 = {
	type = country_event
	title = italy.14.title
	desc = italy.14.desc

	trigger = { }

	fire_only_once = yes

	option = { # adhere to treaty
		name = italy.14.a

		c:PAR = { trigger_event_non_silently = italy.15 }
		c:MOD = { trigger_event_non_silently = italy.15 }

		trigger_event_non_silently = italy.21
	}

	option = { # try keep everything
		name = italy.14.b

		trigger = {
 			global_var:treaty_florence_result >= 2
 		}

		set_global_variable = {
			name = tuscan_approval_number
			value = 0
		}

		if = {
			limit = {
				global_var:treaty_florence_result = 2
			}
			c:MOD = { trigger_event_non_silently = italy.16 }
		}
		else_if = {
			limit = {
				global_var:treaty_florence_result = 3
			}
			c:PAR = { trigger_event_non_silently = italy.16 }
			c:MOD = { trigger_event_non_silently = italy.16 }
		}
	}
}

		}

			limit = {

			}
			set_global_variable = par_florence_conceded

italy.20 = {
	type = country_event
	title = italy.20.title
	desc = italy.20.desc

	trigger = { }

	option = { # accept the result
		name = italy.20.a

		if = {
			limit = {
				OR = {
					AND = {
						global_var:tuscan_approval_number = 2
 						global_var:treaty_florence_result = 3
					}
					AND = {
						global_var:tuscan_approval_number = 1
 						global_var:treaty_florence_result = 2
					}
				}
			}
			trigger_event_non_silently = italy.21
		}
	}

	option = { # war
		name = italy.20.b

		add_casus_belli = { target = scope:from type = casus_belli:cb_annex }
		declare_war_with_cb = { target = scope:from type = casus_belli:cb_annex }
		random_current_war = {
			limit = {
				or = {
					attacker_leader = c:TUS
					defender_leader = scope:from
				}
			}
			save_scope_as = tuscan_war
		}
		c:HAB = {
			join_war_with = {
				target = scope:from
				war = scope:tuscan_war
				call_in_subjects = no
			}
		}
	}
}
